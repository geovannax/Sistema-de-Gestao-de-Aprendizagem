    def get_object(self) -> GroupInvite | None:
    """Retorna o convite com cache em ``self.object`` para evitar múltiplas queries."""
    if hasattr(self, 'object'):
        return self.object

    self.object = (
        GroupInvite.objects
        .select_related('group', 'created_by')
        .filter(
            pk=self.kwargs['invite_pk'],
            group_id=self.kwargs['pk'],
            group__deleted_at__isnull=True,
        )
        .first()
    )
    return self.object

    def post(self, request: HttpRequest, *args: Any, **kwargs: Any) -> HttpResponse:
    """Define ``is_active=False`` e ``expires_at`` para o momento atual."""
    invite = self.get_object()
    invite.is_active = False
    invite.expires_at = timezone.now()
    invite.save(update_fields=['is_active', 'expires_at', 'updated_at'])

    if request.headers.get('HX-Request'):
        return render(
            request,
            'group/partials/_invite_link.html',
            {
                'object': invite.group,
                'invite_expired': True,
            }
        )

    return redirect('group:share', pk=invite.group.pk)
